local function Round(value)
    return math.floor((tonumber(value) or 0) + 0.5)
end

local function Round1(value)
    return tonumber(string.format("%.1f", tonumber(value) or 0)) or 0
end

local function Round2(value)
    return tonumber(string.format("%.2f", tonumber(value) or 0)) or 0
end

local function PulseTypeToValue(v)
    if v == "autocast" then
        return 2
    end
    if v == "button" then
        return 3
    end
    return 1
end

local function ValueToPulseType(v)
    local n = Round(v)
    if n <= 1 then
        return "pixel"
    end
    if n == 2 then
        return "autocast"
    end
    return "button"
end

function Module:ApplyAll()
    local bars = ForgeUI:GetModule("actionBars")
    if bars and bars.RequestLayout then
        bars:RequestLayout()
    end

    local cds = ForgeUI:GetModule("cooldowns")
    if cds and cds.RefreshAll then
        cds:RefreshAll()
    end

    local bags = ForgeUI:GetModule("bags")
    if bags and bags.Refresh then
        bags:Refresh()
    end
end

function Module:BuildPage(contentParent, applyAfterChange)
    local page = CreateFrame("Frame", nil, contentParent)
    page:SetPoint("TOPLEFT", contentParent, "TOPLEFT", 0, 0)
    page:SetWidth(640)
    page:SetHeight(1)

    local y = 16
    local leftX = 20
    local rightX = 330
    local rowGap = 10
    local colWidth = 260
    local headerWidth = 620
    local lane = 1
    local rowHeight = 0

    local builder = {}

    local function PlaceControl(height)
        local x
        local yy = y
        if lane == 1 then
            x = leftX
            lane = 2
        else
            x = rightX
            lane = 1
            y = y + math.max(rowHeight, height) + rowGap
            rowHeight = 0
        end
        rowHeight = math.max(rowHeight, height)
        return x, yy
    end

    local function NextFullWidth(height)
        if lane == 2 then
            lane = 1
            y = y + rowHeight + rowGap
            rowHeight = 0
        end
        local yy = y
        y = y + height + rowGap
        return yy
    end

    function builder:AddHeader(text)
        local h = Skin:CreateSectionHeader(page, text, headerWidth, true)
        local yy = NextFullWidth(34)
        h:SetPoint("TOPLEFT", page, "TOPLEFT", leftX, -yy)
        return h, yy
    end

    function builder:AddSwitch(label, value, onChanged)
        local sw = Skin:CreateSwitch(page, label, value, function(v)
            onChanged(v and true or false)
            if applyAfterChange then
                Module:ApplyAll()
            end
        end, "ON", "OFF")
        local x, yy = PlaceControl(48)
        sw:SetPoint("TOPLEFT", page, "TOPLEFT", x, -(yy + 18))
        if sw.label then
            sw.label:SetWidth(colWidth - 36)
            sw.label:SetJustifyH("LEFT")
            sw.label:SetWordWrap(false)
        end
        return sw
    end

    function builder:AddSlider(label, minV, maxV, step, value, onChanged)
        local x, yy = PlaceControl(48)
        local text = page:CreateFontString(nil, "OVERLAY", "GameFontNormal")
        text:SetPoint("TOPLEFT", page, "TOPLEFT", x, -yy)
        text:SetText(label)

        local slider = Skin:CreateSlider(page, minV, maxV, step, value, function(v)
            onChanged(v)
            if applyAfterChange then
                Module:ApplyAll()
            end
        end, colWidth)
        slider:SetPoint("TOPLEFT", page, "TOPLEFT", x, -(yy + 18))
        return slider
    end

    function builder:AddDropdown(label, items, value, onChanged)
        local x, yy = PlaceControl(48)
        local text = page:CreateFontString(nil, "OVERLAY", "GameFontNormal")
        text:SetPoint("TOPLEFT", page, "TOPLEFT", x, -yy)
        text:SetText(label)

        local dd = Skin:CreateDropdown(page, colWidth, items or {}, value, function(v)
            onChanged(v)
            if applyAfterChange then
                Module:ApplyAll()
            end
        end)
        dd:SetPoint("TOPLEFT", page, "TOPLEFT", x, -(yy + 18))
        return dd
    end

    function builder:AddColorPicker(label, rgba, onChanged)
        local color = {
            tonumber(rgba and rgba[1]) or 1,
            tonumber(rgba and rgba[2]) or 1,
            tonumber(rgba and rgba[3]) or 1,
            tonumber(rgba and rgba[4]) or 1,
        }
        if Skin and Skin.CreateColorPicker then
            local cp = Skin:CreateColorPicker(page, label, color, function(c)
                onChanged(c)
                if applyAfterChange then
                    Module:ApplyAll()
                end
            end, colWidth)
            local x, yy = PlaceControl(34)
            cp:SetPoint("TOPLEFT", page, "TOPLEFT", x, -yy)
            return cp
        end
        return nil
    end

    function builder:AddButton(label, width, onClick)
        local btn = Skin:CreateButton(page, label, math.min(width or colWidth, colWidth), 24)
        local x, yy = PlaceControl(28)
        btn:SetPoint("TOPLEFT", page, "TOPLEFT", x, -yy)
        btn:SetScript("OnClick", onClick)
        return btn
    end

    function builder:Finish()
        if lane == 2 then
            lane = 1
            y = y + rowHeight + rowGap
            rowHeight = 0
        end
        page:SetHeight(y + 24)
        return page:GetHeight()
    end

    return page, builder
end

function Module:SetActiveCategory(key)
    if not self.panel or not self.categoryPages[key] then
        return
    end

    if self.isCategoryAnimating then
        return
    end

    if self.activeCategory == key then
        return
    end

    local oldKey = self.activeCategory
    local oldPage = oldKey and self.categoryPages[oldKey] or nil
    local newPage = self.categoryPages[key]
    local host = self.panel and self.panel.pageHost

    local contentWidth = 480
    if self.panel and self.panel.scroll and self.panel.scroll.GetWidth then
        contentWidth = math.max(420, (self.panel.scroll:GetWidth() or 0) - 22)
    end
    for _, page in pairs(self.categoryPages) do
        page:SetWidth(contentWidth)
    end

    self.activeCategory = key

    if self.panel and self.panel.mainPanel and self.panel.mainPanel.SetBackdropBorderColor then
        local r, g, b = math.random(), math.random(), math.random()
        self.panel.mainPanel:SetBackdropBorderColor(r, g, b, 0.9)
    end

    for id, btn in pairs(self.categoryButtons) do
        if id == key then
            btn:SetBackdropColor(0.18, 0.46, 0.66, 1)
            btn:SetBackdropBorderColor(0.3, 0.7, 1, 1)
        else
            btn:SetBackdropColor(0.12, 0.12, 0.12, 1)
            btn:SetBackdropBorderColor(0.25, 0.25, 0.25, 1)
        end
    end

    if not oldPage or not host then
        for id, page in pairs(self.categoryPages) do
            page:SetShown(id == key)
            if id == key then
                page:ClearAllPoints()
                page:SetPoint("TOPLEFT", host or self.panel.content, "TOPLEFT", 0, 0)
            end
        end
    else
        local oldIndex = self.categoryOrder[oldKey] or 1
        local newIndex = self.categoryOrder[key] or 1
        local dir = (newIndex > oldIndex) and 1 or -1
        local offset = contentWidth

        oldPage:ClearAllPoints()
        oldPage:SetPoint("TOPLEFT", host, "TOPLEFT", 0, 0)
        oldPage:Show()

        newPage:ClearAllPoints()
        newPage:SetPoint("TOPLEFT", host, "TOPLEFT", dir * offset, 0)
        newPage:Show()

        self.isCategoryAnimating = true
        local t = 0
        local duration = 0.18
        host:SetScript("OnUpdate", function(_, elapsed)
            t = t + elapsed
            local p = t / duration
            if p >= 1 then
                p = 1
            end
            p = 1 - ((1 - p) * (1 - p))

            local oldX = -dir * offset * p
            local newX = (dir * offset) * (1 - p)

            oldPage:ClearAllPoints()
            oldPage:SetPoint("TOPLEFT", host, "TOPLEFT", oldX, 0)
            newPage:ClearAllPoints()
            newPage:SetPoint("TOPLEFT", host, "TOPLEFT", newX, 0)

            if t >= duration then
                host:SetScript("OnUpdate", nil)
                oldPage:Hide()
                newPage:ClearAllPoints()
                newPage:SetPoint("TOPLEFT", host, "TOPLEFT", 0, 0)
                self.isCategoryAnimating = false
            end
        end)
    end

    local page = newPage
    self.panel.pageHost:SetHeight(page:GetHeight())
    self.panel.content:SetHeight(page:GetHeight())
    self.panel.scroll:SetVerticalScroll(0)
end

function Module:SetPanelOpen(show)
    if not self.panel then
        return
    end

    local panel = self.panel

    if show then
        if panel.isSlideOpen then
            return
        end
        panel:ClearAllPoints()
        panel:SetPoint("CENTER", UIParent, "CENTER", 0, 0)
        panel.isSlideOpen = true
        panel:Show()
        return
    end

    if not panel.isSlideOpen then
        return
    end
    panel.isSlideOpen = false
    panel:Hide()
end

function Module:TogglePanel()
    if not Skin then
        ForgeUI:Print("ForgeSkin not found.")
        return
    end

    if not self.panel then
        self:CreatePanel()
    end
    if not self.panel then
        return
    end

    self:SetPanelOpen(not self.panel.isSlideOpen)
end

function Module:Initialize()
    self:CreateRightTab()
    self:CreatePanel()
end
